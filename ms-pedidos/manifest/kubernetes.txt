--- 1. INICIAR MINIKUBE ---
minikube version
minikube status
minikube start

--- 2. BUILD Y PUSH IMAGEN ---
cd ms-pedidos
docker build -t 50luisangelsanchezromero/luis-sanchez-ms-pedidos:1.0 .
docker login -u 50luisangelsanchezromero
docker push 50luisangelsanchezromero/luis-sanchez-ms-pedidos:1.0
cd manifest

--- 3. APLICAR MANIFESTS ---
kubectl apply -f luis-sanchez-37-namespace-pedidos.yml
kubectl config set-context --current --namespace=luis-sanchez-37-namespace-pedidos
kubectl apply -f luis-sanchez-37-secret-pedidos.yml
kubectl apply -f luis-sanchez-37-deployment-pedidos.yml
kubectl apply -f luis-sanchez-37-service-pedidos.yml

--- 4. VERIFICAR ---
kubectl get all,secrets -n luis-sanchez-37-namespace-pedidos
kubectl get pods -n luis-sanchez-37-namespace-pedidos
kubectl rollout status deployment/luis-sanchez-37-deployment-pedidos -n luis-sanchez-37-namespace-pedidos
kubectl logs -f deployment/luis-sanchez-37-deployment-pedidos -n luis-sanchez-37-namespace-pedidos

--- 5. PORT-FORWARD ---
kubectl port-forward service/luis-sanchez-37-service-pedidos 8082:8082 -n luis-sanchez-37-namespace-pedidos

--- 6. PROBAR COMUNICACION ENTRE MICROSERVICIOS ---
# GET  http://localhost:8082/api/pedidos
# POST http://localhost:8082/api/pedidos  (consume ms-productos via WebClient)

--- 7. LIMPIAR ---
kubectl delete namespace luis-sanchez-37-namespace-pedidos
minikube delete
